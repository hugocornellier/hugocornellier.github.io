<div className={"tutorial"}>
    <div class="step">
        ## <div class="step-circle">1</div><span class="step-text">Open Settings</span>
    </div>
    <div>
        Tap on the Settings icon to open the settings menu.
    </div>
    <img
        src="/img/tutorials/set_eyes_tut1.png"
        alt="Output Position Step 1"
        class="step-img"
    />

    <div class="step">
        ## <div class="step-circle">2</div><span class="step-text">Open Eye position</span>
    </div>
    <div>
        Locate the <code>Eye position</code> setting under Video Settings and tap <code>Configure</code>.
    </div>
    <img
        src="/img/tutorials/set_eyes_tut2.png"
        alt="Output Position Step 2"
        class="step-img"
    />

    <div class="step">
        ## <div class="step-circle">3</div><span class="step-text">Adjust offsets</span>
    </div>
    <div>
        Drag the horizontal line up or down to adjust the vertical offset. Drag either vertical line left or right to adjust the inter-eye spacing.
    </div>
    <img
        src="/img/tutorials/set_eyes_tut3.png"
        alt="Output Position Step 3"
        class="step-img"
    />

    <div class="step">
        ## <div class="step-circle">4</div><span class="step-text">Save your changes</span>
    </div>
    <div>
        When satisfied, tap the <strong>Save</strong> button at the top right corner.
    </div>
    <img
        src="/img/tutorials/set_eyes_tut4.png"
        alt="Output Position Step 4"
        class="step-img"
    />
</div>
